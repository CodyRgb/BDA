DATASET ANALYSIS USING R

PLATFORM:
RStudio

PROGRAMMING LANGUAGE:
R


PROCEDURE:

1. Open RStudio.

2. Go to:
   File → New File → R Script

3. Copy and paste the complete R program given below.

4. Save the file as:
   dataset_analysis.R

5. Select all the code and click:
   Run

6. The dataset and calculated results will appear in the
   Console.

7. The graphs will appear in the Plots section.


------------------------------------------------------------
R PROGRAM
------------------------------------------------------------

# Create Dataset

student <- data.frame(
  Name = c("Tejas", "Rahul", "Amit", "Priya", "Sneha"),
  Marks = c(75, 82, 68, 90, 85)
)

print(student)


# MINIMUM

print(min(student$Marks))


# MAXIMUM

print(max(student$Marks))


# AVERAGE

print(mean(student$Marks))


# SUM

print(sum(student$Marks))


# COUNT

print(length(student$Marks))


# LINE GRAPH

plot(
  student$Marks,
  type = "o",
  main = "Student Marks - Line Graph",
  xlab = "Student Number",
  ylab = "Marks"
)


# HISTOGRAM

hist(
  student$Marks,
  main = "Marks Histogram",
  xlab = "Marks"
)


# BAR GRAPH

barplot(
  student$Marks,
  names.arg = student$Name,
  main = "Student Marks - Bar Graph",
  xlab = "Students",
  ylab = "Marks"
)


# PIE CHART

pie(
  student$Marks,
  labels = student$Name,
  main = "Student Marks - Pie Chart"
)
