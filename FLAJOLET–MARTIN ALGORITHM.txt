def trailing_zeros(n):
    count = 0

    while n > 0 and n % 2 == 0:
        count += 1
        n //= 2

    return count


data = ["apple", "banana", "mango", "apple", "orange", "banana"]

max_zero = 0

for item in set(data):
    h = sum(ord(c) for c in item)
    zeros = trailing_zeros(h)

    if zeros > max_zero:
        max_zero = zeros

estimate = 2 ** max_zero

print("Data:", data)
print("Estimated distinct elements:", estimate)
