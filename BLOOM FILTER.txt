# Bloom Filter Algorithm

class BloomFilter:
    def __init__(self, size):
        self.size = size
        self.bit_array = [0] * size

    # First hash function
    def hash1(self, item):
        return hash(item) % self.size

    # Second hash function
    def hash2(self, item):
        return (hash(item) * 31 + 7) % self.size

    # Add an item to Bloom Filter
    def add(self, item):
        index1 = self.hash1(item)
        index2 = self.hash2(item)

        self.bit_array[index1] = 1
        self.bit_array[index2] = 1

    # Check whether item may exist
    def check(self, item):
        index1 = self.hash1(item)
        index2 = self.hash2(item)

        if self.bit_array[index1] == 1 and self.bit_array[index2] == 1:
            return True
        else:
            return False


# Create Bloom Filter
bf = BloomFilter(10)

# Insert elements
items = ["apple", "banana", "mango", "orange"]

for item in items:
    bf.add(item)

# Display bit array
print("Bloom Filter:", bf.bit_array)

# Test elements
test_items = ["apple", "banana", "grapes", "orange"]

for item in test_items:
    if bf.check(item):
        print(item, "may be present")
    else:
        print(item, "is definitely not present")
