HIVE PRACTICAL

1. START HIVE

hive


2. CREATE DATABASE

create database bda;


3. SHOW DATABASES

show databases;


4. USE DATABASE

use bda;


5. CREATE TABLE

CREATE TABLE IF NOT EXISTS student (
    id INT,
    name STRING
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE;


6. DESCRIBE TABLE

describe student;


7. LOAD DATA

INSERT INTO TABLE student
VALUES
(1, 'Tejas'),
(2, 'Rahul'),
(3, 'Amit');


8. FETCH ALL ROWS

select * from student;
