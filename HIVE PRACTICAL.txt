1. START HIVE

hive


2. CREATE DATABASE

create database bda;


3. SHOW DATABASES

show databases;


4. USE DATABASE

use bda;


5. CREATE TABLE

create table student(
id INT,
name STRING
)
row format delimited
fields terminated by ',';


6. DESCRIBE TABLE

describe student;


7. LOAD DATA

load data local inpath '/home/cloudera/Desktop/student.csv'
